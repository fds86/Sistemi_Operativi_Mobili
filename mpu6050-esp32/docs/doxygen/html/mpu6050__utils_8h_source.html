46inline void initMPU(Mpu6050* mpu)
47{
48 mpu->i2c_addr = MPU6050_ADDR;
49 Wire.begin(SDA_PIN, SCL_PIN);
50 Wire.beginTransmission(mpu->i2c_addr);
51 Wire.write(0x6B); /* PWR_MGMT_1 register */
